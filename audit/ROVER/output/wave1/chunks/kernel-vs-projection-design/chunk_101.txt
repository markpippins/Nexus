# 🔗 5. Merge semantics (this is where systems usually break)

Identity merge is NOT deletion.

It isreconciliation of meaning.

```
def merge(self, id_a: str, id_b: str, reason: str):    if id_a == id_b:        return    a = self.identity_map.identity_store[id_a]    b = self.identity_map.identity_store[id_b]    # merge aliases    a.aliases.extend(b.aliases)    # unify origin tracking    a.origin_plan_ids.extend(b.origin_plan_ids)    # mark b as deprecated    b.status = "merged"    b.version += 1    # redirect node mappings    for node, idref in self.identity_map.node_to_identity.items():        if idref == id_b:            self.identity_map.node_to_identity[node] = id_a    self.identity_events.append(IdentityEvent(        event_type="MERGE",        source_identity=id_b,        target_identity=id_a,        reason=reason    ))
```
