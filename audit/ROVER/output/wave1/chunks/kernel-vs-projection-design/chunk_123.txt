class KernelEngine:    def __init__(self, state, identity, graph, lineage):        self.state = state        self.identity = identity        self.graph = graph        self.lineage = lineage        self.version = 0    def reduce(self, delta):        # 1. apply receipts (state update placeholder)        self._apply_receipts(delta)        # 2. identity resolution        resolved = self._resolve_identities(delta)        # 3. graph update        self._apply_graph(resolved)        # 4. lineage record        self.lineage.record({            "version": self.version,            "delta_id": delta.delta_id,            "step": "reduce"        })        # 5. commit        self.version += 1        return self.state    # ----------------------    def _apply_receipts(self, delta):        self.state.setdefault("receipts", []).extend(delta.receipts)    def _resolve_identities(self, delta):        return [            self.identity.resolve(r.get("node_id", "unknown"), "")            for r in delta.receipts        ]    def _apply_graph(self, identities):        for i in range(len(identities) - 1):            self.graph.add(GraphEdge(                source=identities[i],                target=identities[i + 1],                relation="derived_from"            ))
```
